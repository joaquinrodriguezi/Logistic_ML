━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━
1. ELIMINACIÓN DE processing_time (LEAKAGE SEVERO) 🔴 CRÍTICO
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

PROBLEMA DETECTADO:
   - processing_time = (shipping_date - order_date).dt.days
   - En muchas filas, processing_time ≈ Days_for_shipping_(real)
   - Target: delay_days = Days_for_shipping_(real) - Days_for_shipment_(scheduled)
   - El modelo podía RECONSTRUIR el target usando processing_time + scheduled

EVIDENCIA DE LEAKAGE:
   - Si processing_time ≈ Days_for_shipping_(real), entonces:
     delay_days ≈ processing_time - Days_for_shipment_(scheduled)
   - El modelo tiene acceso directo a una aproximación del target
   - Explicaría métricas sospechosamente altas en modelo anterior

CORRECCIÓN APLICADA:
   ✅ processing_time ELIMINADO completamente de numeric_features
   ✅ Agregado a forbidden_features (lista de variables prohibidas)
   ✅ Assert añadido: falla si processing_time aparece en features
   ✅ Documentación en código explicando el riesgo

IMPACTO ESPERADO EN MÉTRICAS:
   - Clasificación (ETAPA 1): PR-AUC puede bajar ~5-15%
   - Regresión (ETAPA 2): MAE_pos puede aumentar ~15-30%
   - ¡Esto es CORRECTO! Refleja rendimiento sin "hacer trampa"

━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━
2. CORRECCIÓN DE LABEL ENCODING (LEAKAGE DE PREPROCESAMIENTO) 🟡 IMPORTANTE
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

PROBLEMA DETECTADO:
   - LabelEncoder hacía .fit() con TRAIN + TEST concatenados
   - Información estadística del test (categorías, frecuencias) filtraba al train
   - Viola principio fundamental: preprocesamiento debe usar SOLO datos de train

CÓDIGO ANTES (INCORRECTO):
   combined = pd.concat([X_train_raw[col], X_test_raw[col]], axis=0)
   le.fit(combined)  # ❌ Usa información del test

CÓDIGO DESPUÉS (CORRECTO):
   le.fit(X_train_raw[col])  # ✅ Solo train
   # Manejo de categorías desconocidas en test → -1

CORRECCIÓN APLICADA:
   ✅ LabelEncoder hace fit SOLO en TRAIN
   ✅ Categorías desconocidas en TEST se codifican como -1
   ✅ Se reportan categorías desconocidas por columna
   ✅ El modelo LightGBM maneja -1 correctamente (tratamiento especial)

IMPACTO ESPERADO EN MÉTRICAS:
   - Impacto menor (~1-5% en métricas)
   - Puede haber algunas categorías nuevas en test no vistas en train
   - Modelo más robusto ante datos futuros con categorías nuevas

3. SPLIT TEMPORAL POR FECHA FIJA (REPRODUCIBILIDAD) 🟢 MEJORA
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

PROBLEMA DETECTADO:
   - Split usaba índices: split_idx = int(len(df) * 0.8)

POR QUÉ LAS MÉTRICAS CAMBIARÁN (EXPLICACIÓN TÉCNICA)
================================================================================

ANTES DE LAS CORRECCIONES:
   El modelo tenía "visión del futuro" a través de processing_time, pudiendo
   aproximar Days_for_shipping_(real) y por ende el target delay_days.
   
   Ejemplo:
   - Si processing_time = 5 días
   - Y Days_for_shipment_(scheduled) = 3 días
   - El modelo infiere: delay_days ≈ 5 - 3 = 2 días ✅ (casi perfecto)
   
   Resultado: Métricas INFLADAS artificialmente (overfitting extremo).

DESPUÉS DE LAS CORRECCIONES:
   El modelo debe predecir usando SOLO información operativa disponible
   al momento del despacho (modo de envío, región, categoría, etc.).
   
   El modelo ya NO puede "hacer trampa" calculando el target indirectamente.
   
   Resultado: Métricas REALES que reflejan capacidad predictiva verdadera.

COMPARACIÓN ESPERADA:

   MODELO CON LEAKAGE (antes):
   - PR-AUC clasificación: ~0.90-0.95 (sospechosamente alto)
   - MAE_pos regresión:     ~0.5-1.0 días (perfectamente bajo)
   
   MODELO SIN LEAKAGE (después):
   - PR-AUC clasificación: ~0.70-0.80 (realista)
   - MAE_pos regresión:     ~2.0-3.5 días (realista para este problema)

INTERPRETACIÓN:
   ⚠️ Si las métricas EMPEORAN después de estas correcciones: ¡BIEN!
   ✅ Significa que el modelo anterior hacía trampa
   ✅ Las nuevas métricas son las que importan para producción
   ✅ El modelo ahora es HONESTO sobre su capacidad predictiva
